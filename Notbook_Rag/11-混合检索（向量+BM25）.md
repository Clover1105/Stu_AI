# 一、介绍

## （一）向量检索

直接使用问题 ==> 向量化模型转换为向量 ==> 用这个向量和向量数据库中每一个向量做余弦相似度匹配，返回最相似的几条数据【整个过程，计算的是当前问题向量和文档向量之间的语义关系】

存在的问题：对一些名词的关注度可能不是很高

## （二）BM25检索

BM25检索基于TF-IDF算法的改进的一个算法，它主要关注的是一些名词的检索

`langchain`框架对于BM25 检索也做了集成，但是我们不用，直接手写

- 使用 rank_bm25 库实现 BM25 检索步骤：
    - 第一步：构造 BM25Okapi 检索对象
        - 查询全部的文档数据
        - 分词
        - 基于分词的结果构造对象
    - 第二步：查询问题的得分
        - 把问题用同一种方式分词
        - 把分词结果，送入给 BM25Okapi 检索对象的 get_scores 方法

## （三）rrf算法

因为向量检索出来的结果排名是按照余弦相似度的得分排名的，取值范围是-1~1

BM25检索出来的结果排名是没有限制的，可以任意取值，比如100，102等等

我们最后是把两个检索的结果合并后在送入重排序模型，这个时候就没办法去融合【因为检索结果得分问题】，提出了解决办法：

- 第一种：加权求和【分别给向量检索和BM25检索权重，然后求和，然后排序】，不采纳
- rrf算法【我们只关注文档的排名，不关注文档的分数】，计算公式=1/(k+rank)，其中k是一个参数，默认值60

### 2. 分词

```python
from rank_bm25 import BM25Okapi
import jieba

# 文档集
txts = [。。。。。。]

# 停顿词
stop_words = set([。。。。。。])

# 分词函数
def tokenize(txt):
    t = jieba.cut(txt)
    # strip() 去除空格
    return [i for i in t if i.strip() not in stop_words and len(i.strip()) >= 1]

# 文档分词
tokens = [tokenize(txt) for txt in txts]
print("分词后的文档:")
for i in tokens:
    print(i)

# 创建bm25对象
bm = BM25Okapi(tokens)

# 问题
question = "GPT-5模型是什么？"

# 问题分词
que_tok = tokenize(question)

# 计算bm25得分
scores = bm.get_scores(que_tok)
print("bm25得分:")
for i,j in enumerate(scores):
    if i%5==0:
        print()
    print(f"{j:10.5f}",end="")

# 初始化索引
index = [i for i in range(len(scores))]

# 排序
index.sort(key=lambda x: scores[x], reverse=True)
print(f"\n排序后的索引:{index}")

# 结果
results = []
for i in index[:5]:
    results.append(txts[i])
print(results)
```

# 三、RRF算法

## （一）特殊情况

检索结果能一一对应

```python
# 假设检索结果
# -- 向量
one = [{"id2": 0.95}, {"id9": 0.88}, {"id3": 0.65}, {"id5": 0.75}]
# -- BM25
two = [{"id5": 15.2}, {"id2": 12.1}, {"id9": 0.95}, {"id3": 8.5}]

# RRF算法融合计算top-2
l = []
for i,v_item in enumerate(one):
    for v_key in v_item.keys():
        for j,b_item in enumerate(two):
            for b_key in b_item.keys():
                if v_key == b_key:
                    l.append({
                        v_key: ((1/(60+i))+(1/(60+j)))
                    })
print(l)

# for i in l:
#     print(list(i.values()))

# 排序
l.sort(key=lambda x: list(x.values())[0], reverse=True)
print(l)
```

## （二）一般情况

向量检索结果和BM25检索结果不一定都是一样的
不一样的处理思路：分开处理

```python
# 假设检索结果
# -- 向量
one = [{"id2": 0.95}, {"id9": 0.88}, {"id3": 0.65}, {"id5": 0.75}]
# -- BM25
two = [{"id5": 15.2}, {"id2": 12.1}, {"id9": 0.95}, {"id3": 8.5}]

# RRF算法融合计算top-2
l = []
for i,v_item in enumerate(one):
    for v_key in v_item.keys():
        l.append({
            v_key: (1/(60+i))
        })
for j,b_item in enumerate(two):
    for b_key in b_item.keys():
        f = 0
        for l_item in l:
            if b_key in l_item:
                l_item[b_key] += (1/(60+j))
                f = 1
                break
        if f == 0:
            l.append({
                b_key: (1/(60+j))
            })
print(l)

# for i in l:
#     print(list(i.values()))

# 排序
l.sort(key=lambda x: list(x.values())[0], reverse=True)
print(l)
```

# 四、混合检索

## （一）langchain的api

langchain 提供了 2 个 api：

- `BM25Retriever` ：BM25 的检索器
- `EnsembleRetriever`：混合检索器

流程：

- 先构造出来向量检索器、BM25 的检索器
- 直接把向量检索器对象和BM25 的检索器对象给混合检索器，构造出来 chain
- 使用 chain 来回答问题

```python
from langchain_chroma import Chroma
from langchain_core.documents import Document
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_classic.retrievers import EnsembleRetriever
from langchain_community.retrievers import BM25Retriever

# 测试文档
txts = [。。。。。。]

# 把数据包装为list[Document]格式
docs = [Document(page_content=item,metadata={"id":i}) for i,item in enumerate(txts,start=1)]

# 加载向量化模型
em_model = HuggingFaceEmbeddings(
    model_name=r"G:\models\paraphrase-multilingual-MiniLM-L12-v2",
    model_kwargs={
        "device": "cuda",
        "local_files_only": True,
    }
)

# 将数据存入向量数据库
v_db = Chroma.from_documents(
    documents=docs,
    embedding=em_model,
    persist_directory=r"G:\GitHub\Stu_AI\StuRAG\chromadb_data",
    collection_name="LC_api"
)

# 向量检索器
v_retriever = v_db.as_retriever(search_kwargs={"k": 3})

# BM25检索器
b_retriever = BM25Retriever.from_documents(
    documents=docs,
    k=3
)

# 混合检索器
er_retriever = EnsembleRetriever(
    retrievers=[v_retriever, b_retriever],
)

# 问题
q = "介绍一下GPT-5"

res = er_retriever.invoke(q)
print(res)
```

```python
import jieba
from langchain_core.documents import Document
from rank_bm25 import BM25Okapi

from ai import LoadChroma

stop_words = set([......])

# 分词函数
def tokenize(txts):
    txt = jieba.cut(txts)
    return [t for t in txt if t.strip() not in stop_words and len(t.strip()) >= 1]

# 获取bm25对象和文档内容
def build_bm25_index(vector):
    # 获取所有文档
    docs = vector.get()
    # 取出数据
    ids = docs['ids']
    documents = docs['documents']
    metadatas = docs['metadatas']
    # 包装为list[Document]
    docs = [Document(id=ids[i], page_content=documents[i], metadata={"source": metadatas[i]}) for i in range(len(ids))]
    # 文档分词
    docs_cut = [tokenize(doc) for doc in documents]
    # 创建bm25对象
    bm25 = BM25Okapi(docs_cut)
    return bm25, docs

def bm25_search(bm25, question, docs, k=10):
    # 问题分词
    que_cut = tokenize(question)
    # 获取得分
    scores = bm25.get_scores(que_cut)   # 返回的是一个数组 -- []
    # print(scores)
    # 排序
    sort_scores = sorted(range(len(scores)), key=lambda i: scores[i], reverse=True)[:k]
    # 返回结果
    return [docs[i] for i in sort_scores]

if __name__ == '__main__':
    vector_db = LoadChroma.load_chroma_conn()
    bm25, docs = build_bm25_index(vector_db)
    r = bm25_search(bm25, "公安机关接到家庭暴力报案后应当做什么事情？", docs)
    print(r)
```

```python
def rrf(v_result,b_result):
    scores = {}	# 分数
    docs = {}	# 文档
    
    for index,score in enumerate(v_result,start=1):
        scores[score.id] = scores.get(score.id,0)+round(1/(60+index),4)
        docs[score.id] = score
    for index,doc in enumerate(b_result,start=1):
        scores[doc.id] = scores.get(doc.id,0)+round(1/(60+index),4)
        docs[doc.id] = doc
    
    # 排序 
    sorted_scores = sorted(scores.items(),key=lambda x:x[1],reverse=True)
    # print(sorted_scores)
    
    # 返回文档
    result = [docs[id] for id,score in sorted_scores]
    # print(result)
    return result
```
